49double CCurvasPermeabilidadeChierici::getKrw(double Sw) const {
50 double denominador = 1.0 - _Swirr - _Sor;
51
52 // Proteção de estabilidade estrutural
53 if (denominador <= 1e-6) return 0.0;
54
55 double Swn = (Sw - _Swirr) / denominador;
56
57 // Condições de contorno (Singularidade assintótica prevenida)
58 if (Swn <= 0.0) return 0.0;

60

62 return _krwMax * std::exp(-_Aw * std::pow(Swn, -_Bw));
