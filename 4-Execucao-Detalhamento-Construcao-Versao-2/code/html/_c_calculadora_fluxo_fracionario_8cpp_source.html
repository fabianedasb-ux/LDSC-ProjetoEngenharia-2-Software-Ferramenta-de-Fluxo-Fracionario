59// --- Cálculo do Critério de Rapoport-Leas ---
60double CCalculadoraFluxoFracionario::calcularRapoportLeas(double L, double phi, double sigma) const {
61 if (sigma <= 0 || _k <= 0 || phi <= 0 || _ut <= 0) return 0.0;
62
63 // Obtém endpoint de água avaliando em Sw = 1.0
64 double krw0 = _modeloKr->getKrw(1.0);
65
66 double numerador = L * _ut * _mi_w;
67 double raiz = std::sqrt(phi / _k);
68
69 // Assume-se reservatório wet-water forte (cos(0) = 1.0)
70 double denominador = sigma * krw0 * 1.0;
71
72 return raiz * (numerador / denominador);
73}

75// --- Razão de Mobilidade (M^0) ---
76double CCalculadoraFluxoFracionario::calcularM0() const {
77 double krw0 = _modeloKr->getKrw(1.0);
78 double kro0 = _modeloKr->getKro(0.0);
79
80 if (kro0 <= 0 || _mi_w <= 0) return 0.0;
81
82 return (krw0 / _mi_w) / (kro0 / _mi_o);
83}

85// --- Número de Gravidade (Ng^0) ---
86double CCalculadoraFluxoFracionario::calcularNg0() const {
87 if (std::abs(_ut) < 1e-12 || _mi_o <= 0) return 0.0;
88
89 double kro0 = _modeloKr->getKro(0.0);
90 double delta_rho = _rho_w - _rho_o;
91
92 return (_k * kro0 * delta_rho * _g) / (_ut * _mi_o);
93}

95// --- Núcleo Analítico: Fluxo Fracionário (fw) ---
96double CCalculadoraFluxoFracionario::calcularFw(double sw) const {
97
98 // Clamping termodinâmico para as assíntotas do deslocamento
99 if (sw <= 0.0) return 0.0;
100 if (sw >= 1.0) return 1.0;
101
102 double krw = _modeloKr->getKrw(sw);
103 double kro = _modeloKr->getKro(sw);
104
105 // Condição crítica de estrangulamento de fase (Choke condition)
106 if (krw < 1e-15) return 0.0;
107 if (kro < 1e-15) return 1.0;
108
109 double termo_viscoso = (kro * _mi_w) / (krw * _mi_o);
110 double delta_rho = _rho_w - _rho_o;
111
112 // Termo dependente da angulação topográfica
113 double termo_gravitacional = (_k * kro * delta_rho * _g * std::sin(_angulo)) / (_ut * _mi_o);
114
115 // O retorno pode assumir fw > 1 (contra-fluxo de óleo decantando) em falhas geológicas íngremes
116 return (1.0 - termo_gravitacional) / (1.0 + termo_viscoso);
117}

119// --- Diferenciação Numérica (fw') ---
120double CCalculadoraFluxoFracionario::calcularDerivadaFw(double sw) {
121 // Abordagem de Diferenças Finitas Centrais (h = 10^-5 para mitigar erro de truncamento)
122 double h = 1e-5;
123 double fw_mais = calcularFw(sw + h);
124 double fw_menos = calcularFw(sw - h);
125 return (fw_mais - fw_menos) / (2.0 * h);
126}
