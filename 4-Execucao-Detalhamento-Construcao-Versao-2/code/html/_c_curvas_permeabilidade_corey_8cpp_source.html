30double CCurvasPermeabilidadeCorey::calcularSwNorm(double sw) const {
31 double denominador = 1.0 - _swir - _sor;
32
33 // Tratamento numérico para evitar singularidade e divisão por zero
34 if (denominador <= 1e-6) return 0.0;
35
36 double sw_norm = (sw - _swir) / denominador;
37
38 // Assegura robustez física: Swn nunca será negativo ou maior que a unidade
39 return std::max(0.0, std::min(1.0, sw_norm));
40}
