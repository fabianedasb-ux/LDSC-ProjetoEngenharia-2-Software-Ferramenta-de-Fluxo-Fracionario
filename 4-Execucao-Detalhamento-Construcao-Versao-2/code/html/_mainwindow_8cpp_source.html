330void MainWindow::plotarResultados() {
331 auto calc = simulador->getCalculadora();
332 auto welge = simulador->getWelge();
333
334 double pvi_input = ui->lePVI->text().toDouble();
335 double swi = obterSaturacaoInicialUI();
336 double sor = obterSaturacaoOleoResidualUI();
337 double sw_max = 1.0 - sor;
338
339 // Executa Welge passando os parâmetros reais
340 welge->calcularTangente(calc, swi, sw_max);
341
342 double swFrente = welge->getSwFrente();
343 double vel_choque = welge->getInclinacaoChoque();
344 double t_bt = (vel_choque > 1e-6) ? (1.0 / vel_choque) : 1.0;
345
346 // -----------------------------------------------------------------
347 // 1. PERFIL DE SATURAÇÃO (Sw vs xD)
348 // -----------------------------------------------------------------
349 QVector<double> xD, Sw;
350 double pos_choque = vel_choque * pvi_input;
351
352 xD.append(0.0);
353 Sw.append(sw_max);
354
355 for (double s = sw_max; s >= swFrente; s -= 0.001) {
356 double vel = calc->calcularDerivadaFw(s);
357 double pos = vel * pvi_input;
358
359 if (pos > pos_choque) pos = pos_choque;
360
361 if (pos <= 1.0) {
362 xD.append(pos);
363 Sw.append(s);
364 }
365 }
366
367 if (pos_choque <= 1.0) {
368 xD.append(pos_choque); Sw.append(swi);
369 xD.append(1.0); Sw.append(swi);
370 } else {
371 double target_vel = 1.0 / pvi_input;
372 double sw_out = swFrente;
373 for (double s = sw_max; s >= swFrente; s -= 0.001) {
374 if (calc->calcularDerivadaFw(s) >= target_vel) {
375 sw_out = s;
376 break;
377 }
378 }
379 if (!xD.isEmpty() && xD.last() < 1.0) {
380 xD.append(1.0);
381 Sw.append(sw_out);
382 }
383 }
384
385 ui->plotSvsX->graph(0)->setData(xD, Sw);
386 ui->plotSvsX->xAxis->setRange(0.0, 1.0);
387 ui->plotSvsX->yAxis->setRange(0.0, 1.05);
388
389 // -----------------------------------------------------------------
390 // 2. TANGENTE DE WELGE NO FLUXO FRACIONÁRIO
391 // -----------------------------------------------------------------
392 if (swFrente > swi + 1e-6) {
393 ui->plotFluxo->graph(1)->setData({swi, welge->getSwMedia()}, {calc->calcularFw(swi), 1.0});
394 ui->plotFluxo->rescaleAxes();
395 double yMinF = ui->plotFluxo->yAxis->range().lower;
396 double yMaxF = ui->plotFluxo->yAxis->range().upper;
397 ui->plotFluxo->yAxis->setRange(yMinF - 0.05*(yMaxF-yMinF), yMaxF + 0.05*(yMaxF-yMinF));
398 } else {
399 if (ui->plotFluxo->graphCount() > 1 && ui->plotFluxo->graph(1)->data()) {
400 ui->plotFluxo->graph(1)->data()->clear();
401 }
402 }
403
404 // -----------------------------------------------------------------
405 // 3. EFICIÊNCIA DE DESLOCAMENTO (Ed vs PVI)
406 // -----------------------------------------------------------------
407 std::map<double, double> curvaEd;
408
409 for (double t = 0.0; t <= t_bt; t += 0.01) {
410 curvaEd[t] = t / (1.0 - swi);
411 }
412
413 for (double s = swFrente + 0.001; s <= sw_max; s += 0.001) {
414 double deriv = calc->calcularDerivadaFw(s);
415
416 if (deriv > 1e-6) {
417 double t_saida = 1.0 / deriv;
418
419 if (t_saida >= t_bt && t_saida <= 3.0) {
420 double fw_s = calc->calcularFw(s);
421 double sw_media = s + (1.0 - fw_s) * t_saida;
422 double ed_atual = (sw_media - swi) / (1.0 - swi);
423 curvaEd[t_saida] = ed_atual;
424 }
425 }
426 }
427
428 QVector<double> tPVI, Ed;
429 double max_ed = 0.0;
430
431 for (auto const& par : curvaEd) {
432 double t = par.first;
433 double ed = par.second;
434
435 if (ed >= max_ed) {
436 max_ed = ed;
437 tPVI.append(t);
438 Ed.append(ed);
439 }
440 }
441
442 if (!tPVI.isEmpty() && tPVI.last() < 3.0) {
443 tPVI.append(3.0);
444 Ed.append(Ed.last());
445 }
446
447 ui->plotEfDesl->graph(0)->setData(tPVI, Ed);
448
449 double ed_bt = t_bt / (1.0 - swi);
450 ui->plotEfDesl->graph(1)->setData({t_bt}, {ed_bt});
451
452 ui->plotEfDesl->graph(0)->rescaleAxes();
453 ui->plotEfDesl->xAxis->setRange(0.0, 3.0);
454 double yMinE = ui->plotEfDesl->yAxis->range().lower;
455 double yMaxE = ui->plotEfDesl->yAxis->range().upper;
456 ui->plotEfDesl->yAxis->setRange(std::max(0.0, yMinE - 0.05*(yMaxE-yMinE)), yMaxE + 0.05*(yMaxE-yMinE));
457
458 ui->plotFluxo->replot();
459 ui->plotSvsX->replot();
460 ui->plotEfDesl->replot();
461}

498 try {
499 QString caminho = QFileDialog::getSaveFileName(this, "Salvar Relatório Executivo", "", "PDF (*.pdf)");
500 if (caminho.isEmpty()) return;
501
502 CRelatorio rel;
503
504 // 1. Coleta os parâmetros estáticos
505 double L = ui->leComprimento->text().toDouble();
506 double A = ui->leArea->text().toDouble();
507 double phi = ui->lePorosidade->text().toDouble() / 100.0;
508 double k = ui->lePerm->text().toDouble();
509 double angulo = ui->leAngulo->text().toDouble();
510 double mi_o = ui->leViscOleo->text().toDouble();
511 double mi_w = ui->leViscAgua->text().toDouble();
512
513 rel.gerarCabecalho(L, A, phi, k, angulo, mi_o, mi_w);
514
515 // 2. Coleta dados da calculadora física
516 auto calc = simulador->getCalculadora();
517 double nrl = calc->calcularRapoportLeas(L, phi, 0.03); // Tensão interfacial padrão
518 double M = calc->calcularM0();
519 double Ng = calc->calcularNg0();
520
521 rel.registrarDiagnosticoFisico(nrl, M, Ng);
522
523 // 3. Métricas Avançadas de Welge e Ruptura (BT)
524 auto welge = simulador->getWelge();
525 double swi = obterSaturacaoInicialUI();
526 double swFrente = welge->getSwFrente();
527 double swMedia = welge->getSwMedia();
528 double vel_choque = welge->getInclinacaoChoque();
529 double t_bt = (vel_choque > 1e-6) ? (1.0 / vel_choque) : 1.0;
530
531 // Cálculo analítico da Eficiência de Deslocamento no exato instante de Breakthrough
532 double ed_bt = (swMedia - swi) / (1.0 - swi);
533
534 rel.registrarResultadoWelge(swFrente, swMedia, t_bt, ed_bt);
535
536 // 4. Parâmetros dinâmicos do tempo digitado na interface
537 double pvi_input = ui->lePVI->text().toDouble();
538 // Saturação média atual depende se já passou do BT ou não
539 double sw_media_atual = swMedia;
540 if (pvi_input > t_bt) {
541 double target_deriv = 1.0 / pvi_input;
542 double sor = obterSaturacaoOleoResidualUI();
543 double sw_max = 1.0 - sor;
544 double sw_out = swFrente;
545 for (double s = swFrente; s <= sw_max; s += 0.001) {
546 if (calc->calcularDerivadaFw(s) <= target_deriv) {
547 sw_out = s;
548 break;
549 }
550 }
551 double fw_out = calc->calcularFw(sw_out);
552 sw_media_atual = sw_out + (1.0 - fw_out) * pvi_input;
553 } else {
554 sw_media_atual = swi + pvi_input;
555 }
556 double ed_atual = (sw_media_atual - swi) / (1.0 - swi);
557 if(ed_atual > 1.0 - obterSaturacaoOleoResidualUI() - swi) ed_atual = 1.0 - obterSaturacaoOleoResidualUI() - swi;
558
559 rel.registrarEficiencia(pvi_input, ed_atual);
560
561 // =========================================================
562 // 5. CAPTURA DOS GRÁFICOS DA INTERFACE (MÁGICA VISUAL)
563 // Extrai os Pixmaps com tamanho adequado para o documento A4
564 // =========================================================
565 QPixmap pixFluxo = ui->plotFluxo->toPixmap(500, 350);
566 QPixmap pixSaturacao = ui->plotSvsX->toPixmap(500, 350);
567 QPixmap pixEficiencia = ui->plotEfDesl->toPixmap(500, 350);
568
569 rel.registrarGraficos(pixFluxo, pixSaturacao, pixEficiencia);
570
571 // 6. Exportação definitiva
572 rel.exportarParaPDF(caminho.toStdString());
573 QMessageBox::information(this, "Sucesso", "Relatório PDF gerado com sucesso.");
574
575 } catch (const std::exception& e) {
576 QMessageBox::warning(this, "Erro ao Gerar Relatório", e.what());
577 }
578}
