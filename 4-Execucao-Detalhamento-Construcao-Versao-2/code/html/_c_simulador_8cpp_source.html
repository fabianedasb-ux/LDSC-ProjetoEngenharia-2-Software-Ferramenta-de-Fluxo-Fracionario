101void CSimulador::executarSimulacao(double tempoInjetado, double swi, double sw_max) {
102 if (!_modeloPermeabilidade) {
103 throw std::runtime_error("Exceção Simulador: Impossível realizar varredura sem um algoritmo de curvas definido.");
104 }
105
106 const double tensao_interfacial = 0.03; // N/m típico para sistemas rocha-óleo-água
107
108 // Check-sum analítico: O número de Rapoport-Leas avalia se as forças capilares
109 // estabilizam a onda, ou se o deslocamento requer correção difusiva
110 double nrl = _calculadora->calcularRapoportLeas(_comprimento, _porosidade, tensao_interfacial);
111 if (nrl < 3.0) {
112 std::cerr << "AVISO GEOLÓGICO: Número de Rapoport-Leas (" << nrl << ") < 3.0. A assunção de forças viscosas dominantes pode ser imprecisa neste reservatório." << std::endl;
113 }
114
115 // Sincroniza estado de massa e geometria para a GPU virtual
116 _calculadora->setPropriedades(_mi_w, _mi_o, _rho_w, _rho_o, _k, _angulo, _vazaoInjecao, _area);
117
118 // Delega os esforços resolutivos para a rotina abstrata
119 _solver->calcularPerfilSaturacao(tempoInjetado, swi, sw_max);
120}
