28bool CWelge::calcularTangente(CCalculadoraFluxoFracionario* calc, double swi, double sw_max) {
29
30 // Reset de segurança do estado geométrico
31 _swInicial = swi;
32 _inclinacaoMax = 0.0;
33 _swFrente = swi;
34
35 double fw_swi = calc->calcularFw(swi);
36
37 // Varredura da secante ancorada fortemente em (Swi, fw(Swi))
38 // O passo de discretização (0.001) define a precisão da localização do choque
39 for (double s = swi + 0.001; s <= sw_max; s += 0.001) {
40
41 // Avaliação do coeficiente angular da secante
42 double slope = (calc->calcularFw(s) - fw_swi) / (s - swi);
43
44 // A condição de máximo garante o cumprimento da Condição de Entropia
45 if (slope > _inclinacaoMax) {
46 _inclinacaoMax = slope;
47 _swFrente = s;
48 }
49 }
50
51 // Validação da viabilidade física da injeção (inclinação não nula)
52 if (_inclinacaoMax > 1e-6) {
53
54 // O balanço de materiais de Welge permite encontrar a Saturação Média
55 // através do prolongamento da tangente até o eixo fw = 1.0
56 double fw_frente = calc->calcularFw(_swFrente);
57 _swMedia = _swFrente + (1.0 - fw_frente) / _inclinacaoMax;
58
59 return true;
60 }
61
62 // Falha termodinâmica ou deslocamento impossível
63 _swMedia = swi;
64 return false;
65}
